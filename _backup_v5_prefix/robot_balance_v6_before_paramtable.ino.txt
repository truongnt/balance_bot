/*
 * ============================================================
 *   ROBOT 2 BANH TU CAN BANG - v2 (B-ROBOT EVO2 architecture)
 * ============================================================
 *  Thay doi so voi v1:
 *  1. TIMER1 ISR phat xung step (chinh xac, khong giat boi Serial.print)
 *  2. CASCADED PID: vong ngoai van toc -> dieu chinh target_angle
 *     -> khong can chinh TARGET_ANGLE thu cong, robot khong bi troi
 *  3. ACCELERATION RAMPING: gioi han dSpeed/dt -> motor khong mat buoc,
 *     giam dong peak -> mat me hon
 *  4. AUTO-CALIBRATION goc dung thang: cam robot dung thang khi khoi dong,
 *     code tu lay accel offset -> khong can hardcode TARGET_ANGLE
 *
 *  Phan cung (giong v1):
 *    Arduino Uno + CNC Shield + 2x A4988 + 2x NEMA17 + MPU-6050
 *    Motor trai  (X): STEP=D2, DIR=D5
 *    Motor phai  (Y): STEP=D3, DIR=D6
 *    Enable       : D8 (LOW = bat)
 *    MPU SDA/SCL  : A4 / A5
 *    Microstep 1/8: jumper M0+M1 ON, M2 OFF
 *
 *  Quy trinh tinh chinh:
 *    1. Bat dau Kp_angle=80, Kd_angle=15, Ki_angle=0, Kp_vel=0
 *       -> Chi vong angle hoat dong (giong v1)
 *    2. Tang Kp_angle den khi can bang co ban
 *    3. Tang Kd_angle de giam dao dong
 *    4. BAT vong velocity: Kp_vel=0.05, tang dan
 *       -> Robot se TU bu troi, khong can chinh TARGET_ANGLE_BIAS
 *    5. Neu robot oscillate sau khi bat velocity loop:
 *       -> Doi dau vel_dir (+1 / -1)
 * ============================================================
 */

#include <Wire.h>
#include <EEPROM.h>
#include <avr/interrupt.h>

// ============================================================
// PIN MAPPING (CNC Shield) - dung direct port de toc do
// ============================================================
//  D2 = PORTD bit 2 (STEP_L)
//  D3 = PORTD bit 3 (STEP_R)
//  D5 = PORTD bit 5 (DIR_L)
//  D6 = PORTD bit 6 (DIR_R)
//  D8 = PORTB bit 0 (ENABLE)
#define STEP_L_HIGH()  (PORTD |=  (1 << 2))
#define STEP_L_LOW()   (PORTD &= ~(1 << 2))
#define STEP_R_HIGH()  (PORTD |=  (1 << 3))
#define STEP_R_LOW()   (PORTD &= ~(1 << 3))
#define DIR_L_HIGH()   (PORTD |=  (1 << 5))
#define DIR_L_LOW()    (PORTD &= ~(1 << 5))
#define DIR_R_HIGH()   (PORTD |=  (1 << 6))
#define DIR_R_LOW()    (PORTD &= ~(1 << 6))
#define ENABLE_PIN     8

// MPU-6050
#define MPU_ADDR       0x68

// ============================================================
// THAM SO DIEU KHIEN
// ============================================================

// --- PID vong trong (Angle, ~100Hz) ---
// REVERT v3 config (run 25-giay stable): Kp=80, Kd=12
// Ket hop voi Vref=0.6V: balance 25s + recover ±13°
float Kp_angle = 80.0;
float Ki_angle = 0.0;
float Kd_angle = 12.0;

// --- PID vong ngoai (Velocity, ~10Hz) ---
float Kp_vel  = 0.008;
float Ki_vel  = 0.001;
int8_t vel_dir = -1;             // tune LIVE qua lenh "vdir -1" / "vdir 1"
float max_target_offset = 3.0f;  // tune LIVE qua lenh "mto 3.0"

// Bias cho CG offset - tune LIVE qua lenh "bias +0.5", luu EEPROM bang "save"
float target_bias = 0.0f;

// Velocity filter — tune LIVE qua lenh "vfa 0.6"
float vel_filter_alpha = 0.6f;

// --- Stepper / motor --- tune LIVE qua lenh "maxspd / maxacc"
#define MICROSTEP          8
int16_t max_steps_per_sec = 7000;
int16_t max_accel          = 350;   // B-ROBOT EVO2: MAX_ACCEL 14 x25 (1/8 step) = 350/tick
#define MIN_STEP_RATE      30
#define ZERO_SPEED         32000

// --- An toan --- tune LIVE qua "fall 45 / rec 10 / recg 100"
float fall_angle    = 45.0f;
float recover_angle = 10.0f;
float recover_gyro  = 100.0f;

// --- Kien truc output --- doi LIVE qua lenh "arch 0" / "arch 1"
//  0 = PID -> toc do TRUC TIEP  (ban hien tai)
//  1 = PID -> GIA TOC, tich phan thanh toc do (kieu B-ROBOT EVO2: control_output += ...)
//  LUU Y: hai che do can BO GAIN HOAN TOAN KHAC NHAU.
//    arch 0 -> kp ~80,  kd ~12
//    arch 1 -> kp ~8,   kd ~125   (uoc tu B-ROBOT KP 0.32 / KD 0.050, scale x25)
uint8_t integrate_out  = 0;
float   control_output = 0;      // state cua che do tich phan

// --- MPU DLPF (reg 0x1A) --- tune LIVE qua "dlpf 5"
//  0=260Hz 1=184 2=94 3=44 4=21 5=10 6=5 Hz. Loc manh hon = tre nhieu hon.
uint8_t dlpf_cfg = 0x05;

// --- Vong dieu khien ---
#define LOOP_PERIOD_US     10000UL
#define VEL_LOOP_DIVIDER   10

// --- Complementary filter --- tune LIVE qua lenh "alpha / alfa"
float cf_alpha      = 0.97f;   // Tin gyro bao nhieu (0.95-0.98)
float accel_lpf     = 0.85f;   // Loc noise accel tu motor (0.7-0.9): 0.85 giam filter-drift

// ============================================================
// BIEN TOAN CUC
// ============================================================
uint8_t deferred_print = 0;   // 0=idle, 1-6=in-progress (print 1 dong/loop)

float angle           = 0;     // Goc nghieng filtered (do)
float gyroBias        = 0;     // Offset gyro (raw)
float upright_offset  = 0;     // Offset accel khi robot dung thang (do)

// Inner PID state
float pidI_angle      = 0;

// Outer PID state
float target_angle    = 0;     // Setpoint dong, do velocity loop dieu chinh
float pidI_vel        = 0;
long  last_steps_avg  = 0;

// Speed ramping
int16_t current_speed = 0;     // Toc do hien tai sau ramping (chung cho 2 motor)

// === BLUETOOTH REMOTE CONTROL ===
// Nhan lenh tu HC-05 qua Serial (D0/D1)
float bt_drive    = 0;         // -1.5 .. +1.5 do, them vao target_angle (tien/lui)
int16_t bt_turn   = 0;         // -300 .. +300 step/s, motor R - motor L
#define BT_DRIVE_STEP   0.4f   // Moi lan F/B nhan thay doi target nay
#define BT_DRIVE_MAX    1.5f
#define BT_TURN_STEP    100    // Moi lan L/R nhan thay doi turn nay
#define BT_TURN_MAX     300

// Trang thai
bool fallen           = false;

// Soft-start sau recovery: ramp Kp 50%->100% trong SOFT_START_LOOPS vong
uint16_t soft_start_cnt = 0;
#define SOFT_START_LOOPS  300   // 300 x 10ms = 3 giay

// === Volatile state truy cap trong ISR ===
volatile uint16_t periodL = ZERO_SPEED;   // ticks giua 2 step (ticks=0.5us @ Timer1 prescaler 8)
volatile uint16_t periodR = ZERO_SPEED;
volatile int8_t   dirSignL = 0;            // -1, 0, +1 (cho buoc dem)
volatile int8_t   dirSignR = 0;
volatile long     stepsL = 0;              // Buoc da chay (signed, cho tinh van toc)
volatile long     stepsR = 0;

// Timing
unsigned long last_loop_us = 0;

// ============================================================
// TIMER1 ISR - Phat xung step (precise)
// ============================================================
//  Timer1: 16-bit, prescaler 8 -> 2 MHz tick (0.5us/tick)
//  Dung 2 channel doc lap: OCR1A cho motor L, OCR1B cho motor R
//  Khi compare match: phat 1 xung step (HIGH ngan roi LOW)
//                     va lich next interrupt = OCR + period

ISR(TIMER1_COMPA_vect) {
  uint16_t next = OCR1A + periodL;
  if ((int16_t)(next - TCNT1) < 4) next = TCNT1 + 4;  // OCR slip guard
  OCR1A = next;
  if (periodL >= ZERO_SPEED) return;
  STEP_L_HIGH();
  stepsL += dirSignL;   // volatile long += volatile int8_t: ~25 cycles = 1.5us > A4988 min
  STEP_L_LOW();
}

ISR(TIMER1_COMPB_vect) {
  uint16_t next = OCR1B + periodR;
  if ((int16_t)(next - TCNT1) < 4) next = TCNT1 + 4;  // OCR slip guard
  OCR1B = next;
  if (periodR >= ZERO_SPEED) return;
  STEP_R_HIGH();
  stepsR += dirSignR;
  STEP_R_LOW();
}

void initTimer1() {
  noInterrupts();
  TCCR1A = 0;
  TCCR1B = 0;
  TCNT1  = 0;
  TCCR1B |= (1 << CS11);              // Prescaler 8 -> 2MHz
  OCR1A   = TCNT1 + ZERO_SPEED;
  OCR1B   = TCNT1 + ZERO_SPEED;
  TIFR1   = (1 << OCF1A) | (1 << OCF1B);  // Clear flag
  TIMSK1  = (1 << OCIE1A) | (1 << OCIE1B);
  interrupts();
}

// ============================================================
// SET MOTOR SPEED (voi acceleration ramping + turn support)
// ============================================================
//  base_speed: toc do trung binh, -MAX..+MAX step/s
//  turn_offset: chenh lech R - L (R nhanh hon -> re trai, ngugc lai)
//  Ramping ap dung tren base_speed
void setMotorSpeed(int16_t target_speed) {
  // Ramping tren base_speed
  if (target_speed > current_speed + max_accel)
    current_speed += max_accel;
  else if (target_speed < current_speed - max_accel)
    current_speed -= max_accel;
  else
    current_speed = target_speed;

  // Ap dung turn offset
  int16_t left_speed  = current_speed - bt_turn;
  int16_t right_speed = current_speed + bt_turn;
  setSingleMotor(true,  left_speed);   // motor trai
  setSingleMotor(false, right_speed);  // motor phai
}

// Helper: set 1 motor (left=true cho L, false cho R)
void setSingleMotor(bool isLeft, int16_t speed) {
  int8_t sign;
  uint16_t absspeed;
  if (speed >= 0) { sign = +1; absspeed = speed; }
  else            { sign = -1; absspeed = -speed; }

  // DIR pin (motor R dao chieu vi gan doi xung)
  if (isLeft) {
    if (sign > 0) DIR_L_HIGH(); else DIR_L_LOW();
  } else {
    if (sign > 0) DIR_R_LOW();  else DIR_R_HIGH();
  }

  // Period
  uint16_t newPeriod;
  if (absspeed < MIN_STEP_RATE) {
    newPeriod = ZERO_SPEED;
  } else {
    uint32_t p = 2000000UL / absspeed;
    if (p >= ZERO_SPEED) newPeriod = ZERO_SPEED;
    else                 newPeriod = (uint16_t)p;
  }

  noInterrupts();
  if (isLeft) {
    bool was_stopped = (periodL >= ZERO_SPEED);
    periodL  = newPeriod;
    dirSignL = (newPeriod >= ZERO_SPEED) ? 0 : sign;
    if (was_stopped && newPeriod < ZERO_SPEED) {
      OCR1A = TCNT1 + 100;
    }
  } else {
    bool was_stopped = (periodR >= ZERO_SPEED);
    periodR  = newPeriod;
    dirSignR = (newPeriod >= ZERO_SPEED) ? 0 : sign;
    if (was_stopped && newPeriod < ZERO_SPEED) {
      OCR1B = TCNT1 + 100 + (newPeriod >> 1);
    }
  }
  interrupts();
}

void stopMotors() {
  noInterrupts();
  periodL = ZERO_SPEED;
  periodR = ZERO_SPEED;
  dirSignL = 0;
  dirSignR = 0;
  interrupts();
  current_speed = 0;
}

void enableMotors()  { digitalWrite(ENABLE_PIN, LOW);  }
void disableMotors() { digitalWrite(ENABLE_PIN, HIGH); }

// ============================================================
// MPU-6050
// ============================================================
void writeMPU(uint8_t reg, uint8_t val) {
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(reg); Wire.write(val);
  Wire.endTransmission(true);
}

void readMPU(int16_t *ax, int16_t *ay, int16_t *az,
             int16_t *gx, int16_t *gy, int16_t *gz) {
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x3B);
  Wire.endTransmission(false);
  Wire.requestFrom((uint8_t)MPU_ADDR, (uint8_t)14, (uint8_t)true);
  *ax = (Wire.read() << 8) | Wire.read();
  *ay = (Wire.read() << 8) | Wire.read();
  *az = (Wire.read() << 8) | Wire.read();
  Wire.read(); Wire.read();              // bo qua nhiet do
  *gx = (Wire.read() << 8) | Wire.read();
  *gy = (Wire.read() << 8) | Wire.read();
  *gz = (Wire.read() << 8) | Wire.read();
}

void initMPU() {
  Wire.begin();
  Wire.setClock(400000);
  writeMPU(0x6B, 0x00);   // Wake
  writeMPU(0x1B, 0x00);   // Gyro ±250°/s -> 131 LSB/(°/s)
  writeMPU(0x1C, 0x00);   // Accel ±2g
  writeMPU(0x1A, dlpf_cfg);   // DLPF, mac dinh 10Hz - doi LIVE qua "dlpf n"
                          // Co the giam them 0x05 (10Hz) neu van rung
  delay(100);
}

// Goc tu accelerometer (giu nguyen cong thuc v1, them upright_offset)
float calcAccelAngleRaw(int16_t ay, int16_t az) {
  return atan2f((float)-ay, (float)az) * 57.2957795f;
}

// Calibrate: lay 1000 mau khi robot dung yen
//  - gyroBias: trung binh gx
//  - upright_offset: trung binh accelAngle (== goc bias khi cam dung thang)
void calibrate() {
  Serial.println(F("Calib: GIU robot DUNG THANG va im trong 4 giay..."));
  delay(2500);

  long sumGx = 0;
  float sumAng = 0;
  const int N = 1000;
  for (int i = 0; i < N; i++) {
    int16_t ax, ay, az, gx, gy, gz;
    readMPU(&ax, &ay, &az, &gx, &gy, &gz);
    sumGx  += gx;
    sumAng += calcAccelAngleRaw(ay, az);
    delay(2);
  }
  gyroBias       = (float)sumGx / N;
  upright_offset = sumAng / N;
  Serial.print(F("  gyroBias = "));      Serial.println(gyroBias, 2);
  Serial.print(F("  upright_offset = ")); Serial.println(upright_offset, 2);
}

// ============================================================
// SETUP
// ============================================================
void setup() {
  Serial.begin(115200);
  Serial.println(F("=== ROBOT BALANCE v3 + BT ==="));

  // Load params tu EEPROM (neu co), neu khong dung gia tri mac dinh
  if (loadParams()) {
    Serial.println(F("Params loaded from EEPROM:"));
    printParams();
  } else {
    Serial.println(F("Using default params (no EEPROM data)"));
  }

  // Pin modes
  pinMode(2, OUTPUT); pinMode(3, OUTPUT);
  pinMode(5, OUTPUT); pinMode(6, OUTPUT);
  pinMode(ENABLE_PIN, OUTPUT);
  disableMotors();

  // MPU
  initMPU();
  Serial.println(F("MPU OK"));

  // Calibrate (4 seconds, hold robot upright)
  calibrate();

  // Doc goc khoi tao
  int16_t ax, ay, az, gx, gy, gz;
  readMPU(&ax, &ay, &az, &gx, &gy, &gz);
  angle = calcAccelAngleRaw(ay, az) - upright_offset;
  Serial.print(F("Goc ban dau: ")); Serial.println(angle, 2);
  // BIAS_HINT da BO. Ly do: angle o day = 1 mau don TRU trung binh 1000 mau
  // vua lay xong ngay truoc do -> ky vong toan hoc = 0. No do NHIEU CAM BIEN,
  // khong do CG offset. Gui no thanh BIAS chi lam robot nghieng lech ngau nhien
  // +-1 do moi lan boot. Dung lenh "bias <val>" thu cong neu can bu CG.

  // Khoi tao Timer1 ISR
  initTimer1();

  // Bat motor
  enableMotors();
  delay(100);

  Serial.println(F("Bat dau!"));
  Serial.println(F("angle\taccel\ttargetA\tgyroR\tspeed\tstepsL\tstepsR"));

  last_loop_us = micros();
}

// ============================================================
// BLUETOOTH COMMAND PARSER
// ============================================================
// Lenh 1 ky tu (gui qua app Serial Bluetooth Terminal):
//   F = forward (+target_angle)    B = backward (-target_angle)
//   L = turn left  (R faster)      R = turn right (L faster)
//   S = stop tien/lui/quay         P = print PID gains
// ============================================================
// EEPROM LAYOUT (luu params qua power cycle)
// ============================================================
// MAGIC doi 0xAB42 -> 0xAB43 vi struct them field. Doi MAGIC moi khi doi
// thu tu / them field, neu khong du lieu cu se bi doc lech tung byte.
#define EEPROM_MAGIC   0xAB43
#define EEPROM_ADDR    0

struct Params {
  uint16_t magic;
  float kp_angle, ki_angle, kd_angle;
  float kp_vel, ki_vel;
  float target_bias;
  float cf_alpha, accel_lpf, vel_filter_alpha;
  float max_target_offset;
  int16_t max_steps_per_sec, max_accel;
  float   fall_angle, recover_angle, recover_gyro;
  int8_t  vel_dir;
  uint8_t integrate_out, dlpf_cfg;
};

void saveParams() {
  Params p = { EEPROM_MAGIC,
    Kp_angle, Ki_angle, Kd_angle, Kp_vel, Ki_vel, target_bias,
    cf_alpha, accel_lpf, vel_filter_alpha,
    max_target_offset, max_steps_per_sec, max_accel,
    fall_angle, recover_angle, recover_gyro,
    vel_dir, integrate_out, dlpf_cfg };
  EEPROM.put(EEPROM_ADDR, p);
  Serial.println(F("> SAVED to EEPROM"));
}

bool loadParams() {
  Params p;
  EEPROM.get(EEPROM_ADDR, p);
  if (p.magic != EEPROM_MAGIC) return false;
  Kp_angle         = p.kp_angle;
  Ki_angle         = p.ki_angle;
  Kd_angle         = p.kd_angle;
  Kp_vel           = p.kp_vel;
  Ki_vel           = p.ki_vel;
  target_bias      = p.target_bias;
  cf_alpha         = p.cf_alpha;
  accel_lpf        = p.accel_lpf;
  vel_filter_alpha = p.vel_filter_alpha;
  max_target_offset   = p.max_target_offset;
  max_steps_per_sec   = p.max_steps_per_sec;
  max_accel           = p.max_accel;
  fall_angle          = p.fall_angle;
  recover_angle       = p.recover_angle;
  recover_gyro        = p.recover_gyro;
  vel_dir             = p.vel_dir;
  integrate_out       = p.integrate_out;
  dlpf_cfg            = p.dlpf_cfg;
  return true;
}

void printParams() {
  deferred_print = 1;  // in dan qua cac vong loop, khong block
}

void doPrintStep() {
  switch (deferred_print) {
    case 1: Serial.print(F("Kp="));   Serial.print(Kp_angle,2);
            Serial.print(F(" Ki="));  Serial.print(Ki_angle,3);
            Serial.print(F(" Kd="));  Serial.println(Kd_angle,2); break;
    case 2: Serial.print(F("Kpv=")); Serial.print(Kp_vel,4);
            Serial.print(F(" Kiv=")); Serial.println(Ki_vel,4); break;
    case 3: Serial.print(F("bias=")); Serial.println(target_bias,3); break;
    case 4: Serial.print(F("alpha=")); Serial.print(cf_alpha,3);
            Serial.print(F(" alpf=")); Serial.println(accel_lpf,3); break;
    case 5: Serial.print(F("vfa="));  Serial.print(vel_filter_alpha,3);
            Serial.print(F(" mto=")); Serial.println(max_target_offset,2); break;
    case 6: Serial.print(F("spd="));  Serial.print(max_steps_per_sec);
            Serial.print(F(" acc=")); Serial.println(max_accel); break;
    case 7: Serial.print(F("fall=")); Serial.print(fall_angle,1);
            Serial.print(F(" rec="));  Serial.print(recover_angle,1);
            Serial.print(F(" recg=")); Serial.println(recover_gyro,0); break;
    case 8: Serial.print(F("arch=")); Serial.print(integrate_out);
            Serial.print(F(" dlpf=")); Serial.print(dlpf_cfg);
            Serial.print(F(" vdir=")); Serial.println(vel_dir); break;
  }
  deferred_print++;
  if (deferred_print > 8) deferred_print = 0;
}

// ============================================================
// BLUETOOTH COMMAND PARSER (line-buffered)
// ============================================================
// Protocol: moi lenh la 1 dong ket thuc bang '\n'
// Dieu khien: F B L R S (1 ky tu, khong can newline)
// Tham so:    KP 80.5 | KD 12 | KI 0 | KPV 0.008 | KIV 0.001
//             BIAS 0.5 | BIASADJ +0.3
//             SAVE | LOAD | RESET | GET

void parseCommand(char* buf) {
  // Chuyen sang uppercase
  for (char* p = buf; *p; p++) *p = toupper(*p);

  // Tach keyword va value
  char* sp = strchr(buf, ' ');
  float val = 0;
  if (sp) { *sp = 0; val = atof(sp + 1); }

  // Lenh dieu khien 1 ky tu (gio gui kem \n tu Python)
  if      (strcmp(buf, "F") == 0) { bt_drive = constrain(bt_drive + BT_DRIVE_STEP, -BT_DRIVE_MAX, BT_DRIVE_MAX); pidI_vel = 0; Serial.print(F("> drive=")); Serial.println(bt_drive, 2); }
  else if (strcmp(buf, "B") == 0) { bt_drive = constrain(bt_drive - BT_DRIVE_STEP, -BT_DRIVE_MAX, BT_DRIVE_MAX); pidI_vel = 0; Serial.print(F("> drive=")); Serial.println(bt_drive, 2); }
  else if (strcmp(buf, "L") == 0) { bt_turn = constrain(bt_turn - BT_TURN_STEP, -BT_TURN_MAX, BT_TURN_MAX); Serial.print(F("> turn=")); Serial.println(bt_turn); }
  else if (strcmp(buf, "R") == 0) { bt_turn = constrain(bt_turn + BT_TURN_STEP, -BT_TURN_MAX, BT_TURN_MAX); Serial.print(F("> turn=")); Serial.println(bt_turn); }
  else if (strcmp(buf, "S") == 0) { bt_drive = 0; bt_turn = 0; pidI_vel = 0; Serial.println(F("> STOP")); }
  else if (strcmp(buf, "P") == 0) { printParams(); }
  // Tuning params
  else if (strcmp(buf, "KP") == 0)      { Kp_angle = val; Serial.print(F("> Kp=")); Serial.println(val, 2); }
  else if (strcmp(buf, "KI") == 0) { Ki_angle = val; Serial.print(F("> Ki=")); Serial.println(val, 3); }
  else if (strcmp(buf, "KD") == 0) { Kd_angle = val; Serial.print(F("> Kd=")); Serial.println(val, 2); }
  else if (strcmp(buf, "KPV") == 0){ Kp_vel   = val; Serial.print(F("> Kpv=")); Serial.println(val, 4); }
  else if (strcmp(buf, "KIV") == 0){ Ki_vel   = val; Serial.print(F("> Kiv=")); Serial.println(val, 4); }
  else if (strcmp(buf, "BIAS") == 0) {
    target_bias = val; pidI_vel = 0;
    Serial.print(F("> bias=")); Serial.println(val, 3);
  }
  else if (strcmp(buf, "BIASADJ") == 0) {
    target_bias += val; pidI_vel = 0;
    Serial.print(F("> bias=")); Serial.println(target_bias, 3);
  }
  else if (strcmp(buf, "SAVE") == 0)  {
    if (fallen) {
      saveParams();
    } else {
      Serial.println(F("> SAVE: nghieng robot xuong truoc (EEPROM block 150ms)"));
    }
  }
  else if (strcmp(buf, "LOAD") == 0)  { if (loadParams()) { Serial.println(F("> LOADED")); printParams(); } else Serial.println(F("> No saved data")); }
  else if (strcmp(buf, "ALPHA") == 0) { cf_alpha = constrain(val, 0.8f, 0.99f); Serial.print(F("> alpha=")); Serial.println(cf_alpha, 3); }
  else if (strcmp(buf, "ALPF") == 0)  { accel_lpf = constrain(val, 0.3f, 0.95f); Serial.print(F("> accel_lpf=")); Serial.println(accel_lpf, 3); }
  else if (strcmp(buf, "VFA") == 0)   { vel_filter_alpha = constrain(val, 0.0f, 0.95f); Serial.print(F("> vel_filter=")); Serial.println(vel_filter_alpha, 3); }
  else if (strcmp(buf, "MTO") == 0)   { max_target_offset = constrain(val, 1.0f, 8.0f); Serial.print(F("> max_tgt=")); Serial.println(max_target_offset, 2); }
  else if (strcmp(buf, "SPD") == 0)   { max_steps_per_sec = (int16_t)constrain(val, 500, 7000); Serial.print(F("> max_spd=")); Serial.println(max_steps_per_sec); }
  else if (strcmp(buf, "ACC") == 0)   { max_accel = (int16_t)constrain(val, 100, 3000); Serial.print(F("> max_acc=")); Serial.println(max_accel); }
  // ---- Truoc day phai nap lai firmware moi doi duoc, gio chinh qua BT ----
  else if (strcmp(buf, "DLPF") == 0)  { dlpf_cfg = (uint8_t)constrain(val, 0, 6); writeMPU(0x1A, dlpf_cfg); Serial.print(F("> dlpf=")); Serial.println(dlpf_cfg); }
  else if (strcmp(buf, "FALL") == 0)  { fall_angle    = constrain(val, 15.0f, 80.0f); Serial.print(F("> fall=")); Serial.println(fall_angle, 1); }
  else if (strcmp(buf, "REC") == 0)   { recover_angle = constrain(val, 0.5f, 45.0f);  Serial.print(F("> rec="));  Serial.println(recover_angle, 1); }
  else if (strcmp(buf, "RECG") == 0)  { recover_gyro  = constrain(val, 5.0f, 500.0f); Serial.print(F("> recg=")); Serial.println(recover_gyro, 0); }
  else if (strcmp(buf, "VDIR") == 0)  { vel_dir = (val < 0) ? -1 : 1; pidI_vel = 0; Serial.print(F("> vdir=")); Serial.println(vel_dir); }
  else if (strcmp(buf, "ARCH") == 0)  {
    integrate_out  = (val >= 0.5f) ? 1 : 0;
    control_output = 0; pidI_angle = 0; pidI_vel = 0;   // reset state khi doi che do
    Serial.print(F("> arch=")); Serial.print(integrate_out);
    Serial.println(integrate_out ? F(" TICH PHAN (thu kp 8, kd 125)")
                                 : F(" TRUC TIEP (thu kp 80, kd 12)"));
  }
  else if (strcmp(buf, "CAL") == 0)   {
    if (fallen) { calibrate(); angle = 0; Serial.println(F("> recalibrated")); }
    else        { Serial.println(F("> CAL: nghieng robot xuong truoc (block 4s)")); }
  }
  else if (strcmp(buf, "RESET") == 0) {
    Kp_angle=80; Ki_angle=0; Kd_angle=12; Kp_vel=0.008; Ki_vel=0.001; target_bias=0;
    cf_alpha=0.97f; accel_lpf=0.85f; vel_filter_alpha=0.6f;
    max_target_offset=3.0f; max_steps_per_sec=7000; max_accel=350;
    fall_angle=45.0f; recover_angle=10.0f; recover_gyro=100.0f;
    vel_dir=-1; integrate_out=0; dlpf_cfg=0x05; writeMPU(0x1A, dlpf_cfg);
    pidI_vel=0; pidI_angle=0; control_output=0;
    Serial.println(F("> RESET to defaults"));
  }
  else if (strcmp(buf, "GET") == 0)   { printParams(); }
  else { Serial.print(F("? ")); Serial.println(buf); }
}

void handleBTCommand() {
  static char buf[32];
  static uint8_t pos = 0;

  while (Serial.available() > 0) {
    char c = Serial.read();

    // Tat ca lenh deu buffer den newline (tranh 'B' cua BIAS bi parse la backward)
    if (c == '\n' || c == '\r') {
      if (pos > 0) {
        buf[pos] = 0;
        parseCommand(buf);
        pos = 0;
      }
    } else if (pos < 31) {
      buf[pos++] = c;
    } else {
      pos = 0;  // overflow, reset
    }
  }
}

// ============================================================
// LOOP - 100Hz fixed rate
// ============================================================
uint8_t  vel_counter  = 0;
uint8_t  dbg_counter  = 0;

void loop() {
  // Xu ly lenh BT moi vong (khong block)
  handleBTCommand();

  // === Cho den next loop tick (100Hz) ===
  unsigned long now = micros();
  unsigned long elapsed = now - last_loop_us;
  if (elapsed < LOOP_PERIOD_US) return;
  float dt = elapsed / 1000000.0f;
  last_loop_us = now;

  // In params 1 dong moi TICK (100Hz) - PHAI nam DUOI cong 100Hz.
  // Truoc day no nam TREN cong, ma loop() quay hang nghin lan/10ms
  // -> ca 6 dong xa ra trong vai us -> day TX buffer 64B -> block ~15ms.
  if (deferred_print > 0) doPrintStep();

  // === 1. Doc MPU + Complementary filter ===
  int16_t ax, ay, az, gx, gy, gz;
  readMPU(&ax, &ay, &az, &gx, &gy, &gz);
  float gyro_rate     = -((float)gx - gyroBias) / 131.0f;
  float accel_raw     = calcAccelAngleRaw(ay, az) - upright_offset;
  // Software low-pass tren accel de bot motor vibration noise
  static float accel_angle = 0;
  accel_angle = accel_lpf * accel_angle + (1.0f - accel_lpf) * accel_raw;
  angle = cf_alpha * (angle + gyro_rate * dt) + (1.0f - cf_alpha) * accel_angle;

  // Soft resync: khi CF va accel lech >3° thi keo nhe CF ve phia accel
  // Ngan gyro drift tich luy lau dai ma khong lam nhay cam voi motor noise
  float angle_drift = accel_angle - angle;
  if (fabsf(angle_drift) > 3.0f) {
    angle += 0.04f * angle_drift;
  }

  // === 2. An toan ===
  // Check CA filtered angle VA raw accel angle (de tranh filter drift bypass safety)
  // Neu accel raw bao nghieng > FALL thi 100% robot dang nga, du filter chua catch up
  bool filter_fall = (fabsf(angle) > fall_angle);
  bool accel_fall  = (fabsf(accel_angle) > fall_angle);
  if (filter_fall || accel_fall) {
    if (!fallen) {
      fallen = true;
      stopMotors();
      disableMotors();
      pidI_angle  = 0;
      pidI_vel    = 0;
      target_angle = 0;
      control_output = 0;
      // Neu accel phat hien -> filter da drift, force re-sync angle = accel
      if (accel_fall && !filter_fall) {
        Serial.println(F("[!] FALLEN (filter drift! resync angle)"));
        angle = accel_angle;  // Reset filter ve true angle
      } else {
        Serial.println(F("[!] FALLEN"));
      }
    }
    return;
  }
  if (fallen) {
    if (fabsf(angle) < recover_angle && fabsf(gyro_rate) < recover_gyro) {
      fallen = false;
      noInterrupts(); stepsL = 0; stepsR = 0; interrupts();
      last_steps_avg = 0;
      soft_start_cnt = SOFT_START_LOOPS;  // ramp Kp tu 50% de tranh overshoot post-recovery
      enableMotors();
      Serial.println(F("[+] RECOVERED"));
    } else {
      return;
    }
  }

  // === 3. VONG NGOAI: Velocity PID -> dieu chinh target_angle (10Hz) ===
  if (++vel_counter >= VEL_LOOP_DIVIDER) {
    vel_counter = 0;

    // Doc step counters atomic
    noInterrupts();
    long stepsAvg = (stepsL + stepsR) / 2;
    interrupts();

    // Van toc = delta steps trong 100ms (don vi: steps/100ms)
    long velocity = stepsAvg - last_steps_avg;
    last_steps_avg = stepsAvg;

    // Low-pass filter velocity de bot transient spike
    static float vel_filtered = 0;
    vel_filtered = vel_filter_alpha * vel_filtered
                 + (1.0f - vel_filter_alpha) * (float)velocity;

    // Sai so van toc: muon = 0
    float vel_err = -vel_filtered * vel_dir;
    // Anti-windup: chi tich luy Ki khi can bang va KHONG co lenh drive tu BT
    // Khi co bt_drive -> khong tich luy, tranh integral chong lai lenh dieu khien
    if (fabsf(angle) < 5.0f && fabsf(bt_drive) < 0.01f) {
      pidI_vel += vel_err;
      pidI_vel  = constrain(pidI_vel, -500.0f, 500.0f);
    }

    // bt_drive them vao target_angle de robot tien/lui khi co lenh BT
    target_angle = target_bias + Kp_vel * vel_err + Ki_vel * pidI_vel + bt_drive;
    target_angle = constrain(target_angle, -max_target_offset - 2.0f, max_target_offset + 2.0f);
  }

  // === 4. VONG TRONG: Angle PID (100Hz) ===
  float angle_err = target_angle - angle;
  pidI_angle += angle_err * dt;
  pidI_angle  = constrain(pidI_angle, -200.0f, 200.0f);

  // D-term: dung gyro_rate truc tiep (it nhieu hon sai phan goc)
  // d(error)/dt = -d(angle)/dt = -gyro_rate
  float d_term = -gyro_rate;

  // Soft-start: ramp Kp tu 50%->100% sau recovery, tranh overshoot khi moi phuc hoi
  float kp_eff = Kp_angle;
  if (soft_start_cnt > 0) {
    float t = 1.0f - (float)soft_start_cnt / SOFT_START_LOOPS;
    kp_eff = Kp_angle * (0.5f + 0.5f * t);
    soft_start_cnt--;
  }

  float pid_out = kp_eff * angle_err
                + Ki_angle * pidI_angle
                + Kd_angle * d_term;

  // === 5. Output -> motor (voi acceleration ramping) ===
  if (integrate_out) {
    // arch 1 (B-ROBOT EVO2): pid_out la lenh GIA TOC -> tich phan ra toc do.
    // Con lac nguoc phan ung voi GIA TOC banh, nen day moi la dang chuan.
    control_output += pid_out;
  } else {
    // arch 0: pid_out CHINH LA toc do (ban goc cua du an nay)
    control_output = pid_out;
  }
  control_output = constrain(control_output,
                             -(float)max_steps_per_sec,
                             +(float)max_steps_per_sec);
  setMotorSpeed((int16_t)control_output);

  // === 6. Debug ===
  if (++dbg_counter >= 1) {            // 100Hz print @ 115200 baud (43 bytes x 100Hz = 4300 bytes/s << 11520)
    dbg_counter = 0;
    Serial.print(angle, 2);        Serial.print('\t');
    Serial.print(accel_angle, 2);  Serial.print('\t');  // De theo doi drift filter vs accel
    Serial.print(target_angle, 2); Serial.print('\t');
    Serial.print(gyro_rate, 1);    Serial.print('\t');
    Serial.print(current_speed);   Serial.print('\t');
    noInterrupts();
    long sl = stepsL, sr = stepsR;
    interrupts();
    Serial.print(sl); Serial.print('\t');
    Serial.println(sr);
  }
}
